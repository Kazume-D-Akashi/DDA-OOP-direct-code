
def nth_max_min(lst, n):
    if n > len(lst) or n < 1:
        return "N is out of range"
    
    lst.sort()
    nth_min = lst[n - 1] 
    nth_max = lst[-n]    
    return nth_min, nth_max

arr = [7, 3, 9, 1, 5, 12, 8]
n = 2
result = nth_max_min(arr, n)
print(f"{n}th Min: {result[0]}, {n}th Max: {result[1]}")
